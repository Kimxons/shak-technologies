🎯 Key Principle: Shared services = Maximum reusability across ALL modules

💡 Core Principles

✅ DO

  • Use shared services
  • Follow standard patterns
  • Load dependencies properly
  • Handle success & errors
  • Use normalized responses

❌ DON'T

  • Load services in HTML
  • Duplicate lookup logic
  • Bypass CoreApi
  • Assume success
  • Hard-code URLs

🎯 One Pattern, Many Uses

Shared Services: Used by ALL modules for common operations

Module Services: Specific to each module, but follow same pattern

Result: Uniform, maintainable, reusable code

🔧 Shared Service: CoreApi

The Foundation - ALL API Requests Go Through Here

// ✨ Creates standardized request envelope
const envelope = CoreApi.makeRequestEnvelope(
    "FormID",           // Form identifier
    { data: "here" },   // Your request data
    "APP_NAME"          // Optional app name
);

// ✨ Makes API call and normalizes response
const result = await CoreApi.post(endpoint, envelope);

if (result.success) {
    console.log(result.data);    // ✅ Success data
} else {
    console.log(result.message); // ❌ Error message
}
💡 Why CoreApi?
  • Automatically creates request envelope
  • Normalizes ALL responses (success/error)
  • One place to update for all modules
  • Consistent error handling everywhere

🔍 Shared Service: SearchService

Generic Search - Works for Any Table

// ✨ Search any table with standard parameters
const result = await SearchService.search({
    TableID: "clientId",              // Table to search

🚀 Create New Module - Step 1

1 Create Service File

// File: services/loans/loanService.js
(function (global) {
  const CoreApi = global.CoreApi;
  const Environment = global.Environment || {};

  const BASE_URL = (Environment.baseUrlLoans || "http://localhost:6903")
                   .replace(/\/+$/, "");

  const LoanService = {
    // ✨ Get loan details
    getLoan(requestData) {
      const envelope = CoreApi.makeRequestEnvelope("GetLoan", requestData);
      return CoreApi.post(`${BASE_URL}/api/v1/Loans/GetLoan`, envelope);
    },

    // ✨ Create new loan
    createLoan(requestData) {
      const envelope = CoreApi.makeRequestEnvelope("CreateLoan", requestData);
      return CoreApi.post(`${BASE_URL}/api/v1/Loans/CreateLoan`, envelope);
    },

    // ✨ Update loan
    updateLoan(requestData) {
      const envelope = CoreApi.makeRequestEnvelope("UpdateLoan", requestData);
      return CoreApi.post(`${BASE_URL}/api/v1/Loans/UpdateLoan`, envelope);
    }
  };

  global.LoanService = LoanService;
})(window);

🚀 Create New Module - Step 3

3 Create Page JavaScript

// File: pages/loans/loan-maintenance.js
(async function() {
  const { ServiceLoader } = window;
  let dependenciesReady = false;
  
  // 1️⃣ Load all dependencies
  try {
    await ServiceLoader.loadCore();          // Core API
    await ServiceLoader.loadLoanService();   // Your module
    await ServiceLoader.loadLookupService(); // Dropdowns
    await ServiceLoader.loadSearchService(); // Search
    dependenciesReady = true;
  } catch (error) {
    console.error("Failed to load dependencies:", error);
    return;
  }

  // 2️⃣ Use the services
  const LoanService = window.LoanService;
  const LookupService = window.LookupService;
  
  // 3️⃣ Initialize page
  function initPage() {
    if (!dependenciesReady) {
      setTimeout(initPage, 100);
      return;
    }
    // Your page logic here
  }
  
  initPage();
})();

📝 Common Operation: Dropdowns

// ✨ Initialize all dropdowns on page load
async function initializeDropdowns() {
    const LookupService = window.LookupService;
    
    // Client Type dropdown
    const clientTypes = await LookupService.getSystemCodeOptions("ClientTypeID");
    populateSelect("clientType", clientTypes);
    
    // Title dropdown
    const titles = await LookupService.getSystemCodeOptions("TitleID");
    populateSelect("title", titles);
    
    // Gender dropdown
    const genders = await LookupService.getSystemCodeOptions("GenderID");
    populateSelect("gender", genders);
    
    // Country dropdown
    const countries = await LookupService.getSystemCodeOptions("CountryID");
    populateSelect("country", countries);
}

function populateSelect(elementId, options) {
    const select = document.getElementById(elementId);
    select.innerHTML = '<option value="">Select...</option>';
    options.forEach(opt => {
        select.innerHTML += `<option value="${opt.value}">${opt.label}</option>`;
    });
}
⚡ Pro Tip: LookupService caches results, so you can call it multiple times without performance impact!

🔍 Common Operation: Search

// ✨ Search function (works for ANY table)
async function searchRecords(searchTerm, tableId) {
    const SearchService = window.SearchService;
    
    const result = await SearchService.search({
        TableID: tableId,  // "clientId", "loanId", "accountId", etc.
        WhereStmt: searchTerm ? `${tableId} like '%${searchTerm}%'` : "",
        OrderBy: `order by ${tableId} asc`,
        PrevOrNext: "1",
        RefID: "",
        OperatorID: getCurrentUser(),
        ModuleID: getCurrentModuleId(),
        OurBranchID: getCurrentBranch()
    });
    
    if (result.success) {
        displaySearchResults(result.data);
    } else {
        showError(result.message);
    }
}

// Usage examples:
searchRecords("12345", "clientId");   // Search clients
searchRecords("LN001", "loanId");     // Search loans
searchRecords("ACC", "accountId");    // Search accounts

📥 Common Operation: GET (Read)

// ✨ Get a single record by ID
async function loadRecord(id) {
    const YourService = window.YourService;
    
    const result = await YourService.getData({
        ID: id,
        // Other parameters as needed
    });
    
    if (result.success) {
        // ✅ Success - use result.data
        const record = result.data;
        populateForm(record);
        showToast("Record loaded successfully", "success");
    } else {
        // ❌ Error - show result.message
        showToast(result.message, "danger");
        console.error("Error:", result.code, result.message);
    }
}

// Example: Load client
async function loadClient(clientId) {
    const result = await ClientService.getClient({ ClientID: clientId });
    if (result.success) {
        populateForm(result.data);
    }
}
💡 Always check result.success before using result.data

📤 Common Operation: POST (Create)

// ✨ Create a new record
async function createRecord() {
    const YourService = window.YourService;
    const formData = collectFormData(); // Get data from form
    
    const result = await YourService.create(formData);
    
    if (result.success) {
        // ✅ Success
        showToast("Record created successfully!", "success");
        clearForm();
        
        // Optional: Get created record ID from response
        const newId = result.data?.ID || result.data?.ClientID;
        console.log("Created record ID:", newId);
    } else {
        // ❌ Error
        showToast(result.message, "danger");
    }
}

// Example: Create client

        ClientTypeID: document.getElementById("clientType").value,
        // ... other fields

🔄 Common Operation: PUT (Update)

// ✨ Update existing record
async function updateRecord(id) {
    const YourService = window.YourService;
    const formData = collectFormData();
    
    // Include ID in the data
    formData.ID = id;
    
    const result = await YourService.update(formData);
    
    if (result.success) {
        // ✅ Success
        showToast("Record updated successfully!", "success");
    } else {
        // ❌ Error
        showToast(result.message, "danger");
    }
}

// ✨ Smart save function (create OR update)
async function saveRecord() {
    const id = document.getElementById("recordId").value;
    
    if (id) {
        // Has ID = Update
        await updateRecord(id);
    } else {
        // No ID = Create
        await createRecord();
    }
}

📋 Standard Request Format

ALL API Requests Follow This Structure

{
    "RequestID": "FormID_1737191234567",
    "FormID": "p_YourFormID",
    "RequestData": {
        "Field1": "value1",
        "Field2": "value2",
        "Field3": "value3"
    },
    "RequestTime": "2026-01-18T10:30:45",
    "AppName": "CORE_BANKING",
    "Checksum": ""
}

Request Fields

  • RequestID: Auto-generated (FormID_timestamp)
  • FormID: Identifies the operation
  • RequestData: Your actual data
  • RequestTime: ISO format timestamp
  • AppName: From environment config

✨ CoreApi Does This

You don't build this manually!

const envelope = 
  CoreApi.makeRequestEnvelope(
    "FormID",
    { your: "data" }
  );

CoreApi creates the envelope automatically!

✅ Standard Response Format

ALL API Responses Are Normalized

{
    "success": true,
    "code": "00",
    "message": "Operation completed successfully",
    "data": {
        "ClientID": "12345",
        "ClientName": "John Doe",
        "ClientTypeID": "01"
    }
}
Field Type Description
success boolean true = success, false = error
code string "00" = success, others = error codes
message string Human-readable message
data object Response payload (null on error)
✨ CoreApi Normalizes: No matter what the backend returns, you always get this consistent format!

⭐ Best Practices

✅ DO

  • Use shared services for common operations
  • Follow the established patterns
  • Load dependencies at page start
  • Check result.success always
  • Use normalized responses
  • Let CoreApi handle requests
  • Cache lookups (auto-cached)
  • Keep HTML minimal (2 scripts)

❌ DON'T

  • Load services directly in HTML
  • Create duplicate lookup logic
  • Bypass CoreApi for API calls
  • Assume responses are successful
  • Hard-code URLs (use Environment)
  • Build request envelopes manually
  • Create custom response formats
  • Load unnecessary dependencies
🎯 Golden Rule: If multiple modules need it, put it in shared/

🔧 Troubleshooting

❌ Services not loading?

  • Check browser console for 404 errors
  • Verify paths in serviceLoader.js
  • Ensure environment.js is loaded first
  • Check file names match exactly (case-sensitive)

❌ Dropdowns empty?

  • Verify CodeID matches backend system codes
  • Check baseUrlSystemCodes in environment.js
  • Check network tab for API errors
  • Ensure LookupService is loaded before calling

❌ Search not working?

  • Verify TableID matches backend table name
  • Check ModuleID and OurBranchID values
  • Ensure OrderBy has "order by" prefix
  • Check WhereStmt syntax (SQL syntax)

❌ API calls failing?

  • Check FormID matches backend endpoint
  • Verify base URL in environment.js
  • Check RequestData has required fields
  • Look at response.message for details

🌍 Real World Example

Creating a Loan Application Module

// 1️⃣ Create loanService.js
const LoanService = {
    getLoan(data) {
        return CoreApi.post(ENDPOINT, 
            CoreApi.makeRequestEnvelope("GetLoan", data));
    },
    createLoan(data) {
        return CoreApi.post(ENDPOINT, 
            CoreApi.makeRequestEnvelope("CreateLoan", data));
    }
};

// 2️⃣ In your page JS
async function initLoanPage() {
    await ServiceLoader.loadLoanService();
    await ServiceLoader.loadLookupService();
    
    // Populate loan type dropdown
    const loanTypes = await LookupService.getSystemCodeOptions("LoanTypeID");
    populateDropdown("loanType", loanTypes);
    
    // Search for client
    document.getElementById("searchBtn").onclick = async () => {
        const result = await SearchService.search({
            TableID: "clientId",
            WhereStmt: "clientId like '%search%'"
        });
        displayResults(result.data);
    };
    
    // Save loan
    document.getElementById("saveBtn").onclick = async () => {
        const result = await LoanService.createLoan(getFormData());
        if (result.success) alert("Loan created!");
    };
}
