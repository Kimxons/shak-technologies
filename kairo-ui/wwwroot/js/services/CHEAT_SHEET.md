# Service Layer Cheat Sheet

## 🚀 Creating a New Module in 5 Steps

### 1. Create Service File
```javascript
// services/yourModule/yourModuleService.js
(function (global) {
  const CoreApi = global.CoreApi;
  const BASE_URL = (Environment.baseUrlCommon || "http://localhost:XXXX").replace(/\/+$/, "");
  
  const YourModuleService = {
    getData(requestData) {
      const envelope = CoreApi.makeRequestEnvelope("GetData", requestData);
      return CoreApi.post(`${BASE_URL}/api/OldAPI`, envelope);
    }
  };
  
  global.YourModuleService = YourModuleService;
})(window);
```

### 2. Register in ServiceLoader
```javascript
// services/shared/serviceLoader.js
async function loadYourModuleService() {
  await loadCore();
  return loadScript(`${getBasePath()}/services/yourModule/yourModuleService.js`);
}

global.ServiceLoader = {
  // ...
  loadYourModuleService,
};
```

### 3. Create Page JS
```javascript
// pages/yourModule/your-page.js
(async function() {
  const { ServiceLoader } = window;
  
  await ServiceLoader.loadCore();
  await ServiceLoader.loadYourModuleService();
  // If using dropdowns, load ONE of these depending on where the codes live:
  await ServiceLoader.loadLookupService();  // Standard System Codes
  // await ServiceLoader.loadCustomCodesLookupService(); // Custom dropdown codes
  // await ServiceLoader.loadCustomCodesLookUPSERCH(); // Legacy alias
  await ServiceLoader.loadSearchService();   // If using search
  
  // Your page logic here
})();
```

### 4. Create HTML
```html
<script src="../../assets/js/services/shared/serviceLoader.js"></script>
<script src="../../assets/js/pages/yourModule/your-page.js"></script>
```

### 5. Add to Environment
```javascript
// environment.js
const Environment = {
  baseUrlYourModule: "http://localhost:XXXX",
};
```

## 📚 Common Operations

### Dropdowns (System Codes)
```javascript
const LookupService = window.LookupService;
const options = await LookupService.getSystemCodeOptions("CodeID");
// Returns: [{ value, label, order }, ...]

// Populate select
const select = document.getElementById("mySelect");
select.innerHTML = '<option value="">Select...</option>';
options.forEach(opt => {
  select.innerHTML += `<option value="${opt.value}">${opt.label}</option>`;
});
```

### Dropdowns (Custom Codes)
Use this when the codes you need are not present in the standard System Codes table.

```javascript
await ServiceLoader.loadCustomCodesLookupService();

const customCodesLookupService = window.customCodesLookupService;
const options = await customCodesLookupService.getCustomCodeOptions("DepreciationRateID");
// Returns: [{ value, label, order }, ...]
```

Direct call (if you need the raw normalized response):

```javascript
const resp = await customCodesLookupService.getCustomDropDownCodes({
  CodeID: "DepreciationRateID"
});
if (resp.success) {
  console.log(resp.data); // usually the array of rows (Details)
}
```

### Search
```javascript
const SearchService = window.SearchService;
const result = await SearchService.search({
  TableID: "yourTable",
  WhereStmt: "id like '%term%'", 
  PrevOrNext: "1",
  RefID: "",
  OperatorID: "web_portal",
  ModuleID: 1000,
  OurBranchID: "002"
});

if (result.success) {
  displayResults(result.data);
}
```

### GET Request
```javascript
const result = await YourService.getData({ ID: "123" });
if (result.success) {
  console.log(result.data);
} else {
  console.error(result.message);
}
```

### POST/Create
```javascript
const result = await YourService.create({
  Field1: "value",
  Field2: "value"
});

if (result.success) {
  alert("Created successfully!");
} else {
  alert(result.message);
}
```

### PUT/Update
```javascript
const result = await YourService.update({
  ID: "123",
  Field1: "new value"
});

if (result.success) {
  alert("Updated successfully!");
}
```

### DELETE
```javascript
const result = await YourService.delete({ ID: "123" });
if (result.success) {
  alert("Deleted successfully!");
}
```

## 🎯 Request Format (Auto-Generated by CoreApi)
```javascript
{
  RequestID: "FormID_1234567890",
  FormID: "p_YourFormID",
  RequestData: { ...your data },
  RequestTime: "2026-01-17T10:00:00",
  AppName: "YOUR_APP",
  Checksum: ""
}
```

**You don't create this manually!** Use:
```javascript
const envelope = CoreApi.makeRequestEnvelope("FormID", { your: "data" });
```

---

## ✅ Response Format (Auto-Normalized)
```javascript
{
  success: true,        // or false
  code: "00",          // "00" = success, others = error codes
  message: "Success",  // Human-readable message
  data: { ... }        // Response payload
}
```

**Always check `result.success` before using `result.data`!**

## 💡 Quick Tips

### HTML Structure
```html
<!-- ONLY 2 scripts needed! -->
<script src="../../assets/js/services/shared/serviceLoader.js"></script>
<script src="../../assets/js/pages/yourModule/your-page.js"></script>
```

### Page JS Structure
```javascript
(async function() {
  // 1. Load dependencies
  await ServiceLoader.loadCore();
  await ServiceLoader.loadYourService();
  
  // 2. Use services
  const result = await YourService.getData();
  
  // 3. Initialize page
  function initPage() {
    // Your logic
  }
  initPage();
})();
```

### Service Structure
```javascript
(function (global) {
  const CoreApi = global.CoreApi;
  const BASE_URL = Environment.baseUrlYourModule;
  
  const YourService = {
    methodName(requestData) {
      const envelope = CoreApi.makeRequestEnvelope("FormID", requestData);
      return CoreApi.post(`${BASE_URL}/endpoint`, envelope);
    }
  };
  
  global.YourService = YourService;
})(window);
```

## 🔧 Troubleshooting

### Services not loading?
- Check browser console for 404 errors
- Verify paths in serviceLoader.js
- Ensure environment.js is loaded
- Check file names (case-sensitive)

### Dropdowns empty?
- Verify CodeID matches backend
- Check baseUrlSystemCodes in environment.js
- If the code is not in System Codes, use `customCodesLookupService` (legacy alias: `customCodesLookUPSERCH`)
- Check network tab for API errors
- Ensure LookupService loaded before calling

### Search not working?
- Verify TableID matches backend table
- Check ModuleID and OurBranchID
- Ensure OrderBy has "order by" prefix
- Check WhereStmt SQL syntax

### API calls failing?
- Check FormID matches backend endpoint
- Verify base URL in environment.js
- Ensure RequestData has required fields
- Look at result.message for error details

## ⭐ Best Practices

### ✅ DO
- Use shared services for common operations
- Follow established patterns
- Load dependencies at page start
- Check `result.success` always
- Use normalized responses
- Let CoreApi handle requests

### ❌ DON'T
- Load services directly in HTML
- Create duplicate lookup logic
- Bypass CoreApi for API calls
- Assume responses are successful
- Hard-code URLs (use Environment)
- Build request envelopes manually

## 🚀 Quick Start Example

```javascript
// pages/yourModule/your-page.js
(async function() {
  const { ServiceLoader } = window;
  
  // Load everything
  await ServiceLoader.loadCore();
  await ServiceLoader.loadYourService();
  await ServiceLoader.loadLookupService();
  await ServiceLoader.loadSearchService();
  
  // Get services
  const YourService = window.YourService;
  const LookupService = window.LookupService;
  const SearchService = window.SearchService;
  
  // Populate dropdowns
  const types = await LookupService.getSystemCodeOptions("TypeID");
  populateDropdown("typeSelect", types);
  
  // Search button
  document.getElementById("searchBtn").onclick = async () => {
    const result = await SearchService.search({
      TableID: "yourTable",
      WhereStmt: "id like '%search%'",
      OrderBy: "order by id asc"
    });
    if (result.success) displayResults(result.data);
  };
  
  // Save button
  document.getElementById("saveBtn").onclick = async () => {
    const data = getFormData();
    const result = await YourService.create(data);
    if (result.success) alert("Saved!");
  };
})();
```
